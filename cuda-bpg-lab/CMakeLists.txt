cmake_minimum_required(VERSION 3.24)
project(cuda_bpg_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BPG_ENABLE_CUDA "Build the CUDA experiments" ON)
set(BPG_LABS "01_measurement;02_global_memory;03_memory_hierarchy;04_execution_config;05_instructions_control;06_async_execution;07_gemm" CACHE STRING "Semicolon-separated implemented labs")
include(CTest)

if(BUILD_TESTING)
  add_executable(host_tests tests/host_tests.cpp)
  target_include_directories(host_tests PRIVATE "${PROJECT_SOURCE_DIR}")
  add_test(NAME host_tools COMMAND host_tests)
  set_tests_properties(host_tools PROPERTIES LABELS host)
endif()

if(BPG_ENABLE_CUDA)
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    message(FATAL_ERROR "Specify -DCMAKE_CUDA_ARCHITECTURES=XX (actual GPU, SM70-SM89). For CPU tools only: -DBPG_ENABLE_CUDA=OFF")
  endif()
  foreach(arch IN LISTS CMAKE_CUDA_ARCHITECTURES)
    if(NOT arch MATCHES "^[78][0-9](-real|-virtual)?$")
      message(FATAL_ERROR "Only explicit SM70-SM89 targets are in scope: ${arch}")
    endif()
  endforeach()
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "CUDA Toolkit not found. Use -DBPG_ENABLE_CUDA=OFF for host tests; GPU code is Not Tested.")
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  include(CheckSourceCompiles)
  check_source_compiles(CUDA "__global__ void __maxnreg__(32) probe() {}\nint main() { return 0; }" BPG_HAS_MAXNREG)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  endif()
  set(BPG_NVCC_OPTIONS -O3 -lineinfo --ptxas-options=-v)
  string(TOUPPER "${CMAKE_BUILD_TYPE}" BPG_BUILD_TYPE_UPPER)
  set(BPG_CONFIG_FLAGS "${CMAKE_CUDA_FLAGS_${BPG_BUILD_TYPE_UPPER}}")
  configure_file(common/build_info.hpp.in generated/bpg_build_info.hpp @ONLY)
  foreach(lab IN LISTS BPG_LABS)
    if(NOT lab MATCHES "^(01_measurement|02_global_memory|03_memory_hierarchy|04_execution_config|05_instructions_control|06_async_execution|07_gemm)$")
      message(FATAL_ERROR "${lab} is planned, not implemented")
    endif()
    add_subdirectory(labs/${lab})
  endforeach()
endif()
