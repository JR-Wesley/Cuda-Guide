add_executable(gemm gemm.cu)
target_include_directories(gemm PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/generated")
target_compile_options(gemm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${BPG_NVCC_OPTIONS}>)
set_target_properties(gemm PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
if(BUILD_TESTING)
  add_test(NAME gemm_help COMMAND gemm --help)
  set_tests_properties(gemm_help PROPERTIES LABELS host)
  foreach(shape "1,1,1" "1,33,17" "33,1,17" "17,31,15" "37,65,33" "65,37,17" "128,128,128" "16,16,1" "1,1,4096")
    string(REPLACE "," ";" dims "${shape}")
    list(GET dims 0 m)
    list(GET dims 1 n)
    list(GET dims 2 k)
    foreach(tile 16 32)
      math(EXPR threads "${tile} * ${tile}")
      foreach(reg_rows 2 4)
        add_test(NAME gemm_${m}_${n}_${k}_t${tile}_r${reg_rows} COMMAND gemm --rows ${m} --cols ${n} --depth ${k} --tile ${tile} --block ${threads} --reg-rows ${reg_rows} --warmup 1 --repeats 2 --iterations 2)
        set_tests_properties(gemm_${m}_${n}_${k}_t${tile}_r${reg_rows} PROPERTIES LABELS "gpu;gemm")
      endforeach()
    endforeach()
  endforeach()
  foreach(pattern identity cancellation)
    foreach(k 31 32)
      foreach(reg_rows 2 4)
        add_test(NAME gemm_${pattern}_${k}_r${reg_rows} COMMAND gemm --rows 37 --cols 65 --depth ${k} --reg-rows ${reg_rows} --pattern ${pattern} --warmup 1 --repeats 2 --iterations 2)
        set_tests_properties(gemm_${pattern}_${k}_r${reg_rows} PROPERTIES LABELS "gpu;gemm")
      endforeach()
    endforeach()
  endforeach()
endif()
