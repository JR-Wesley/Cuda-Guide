add_executable(global_copy global_copy.cu)
target_include_directories(global_copy PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/generated")
target_compile_options(global_copy PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${BPG_NVCC_OPTIONS}>)
set_target_properties(global_copy PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
if(BUILD_TESTING)
  add_test(NAME global_copy_help COMMAND global_copy --help)
  set_tests_properties(global_copy_help PROPERTIES LABELS host)
  foreach(n 1 2 3 4 5 31 32 33 255 256 257 4099)
    add_test(NAME global_copy_n${n} COMMAND global_copy --n ${n} --warmup 1 --repeats 2 --iterations 3)
    set_tests_properties(global_copy_n${n} PROPERTIES LABELS "gpu;global_copy")
  endforeach()
  foreach(direction gather scatter both)
    foreach(stride 1 2 3 8 32)
      add_test(NAME global_copy_${direction}_s${stride} COMMAND global_copy --variant scalar --direction ${direction} --stride ${stride} --offset 3 --n 4099 --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(global_copy_${direction}_s${stride} PROPERTIES LABELS "gpu;global_copy")
    endforeach()
  endforeach()
endif()
