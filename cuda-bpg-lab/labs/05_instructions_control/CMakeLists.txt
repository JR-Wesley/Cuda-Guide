foreach(experiment integer_division loop_unroll branch_flow)
  add_executable(${experiment} ${experiment}.cu)
  target_include_directories(${experiment} PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/generated")
  target_compile_options(${experiment} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${BPG_NVCC_OPTIONS}>)
  set_target_properties(${experiment} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
  if(BUILD_TESTING)
    add_test(NAME ${experiment}_help COMMAND ${experiment} --help)
    set_tests_properties(${experiment}_help PROPERTIES LABELS host)
  endif()
endforeach()
if(BUILD_TESTING)
  foreach(config "1,1,7" "9,1,8" "33,3,7" "4099,31,8" "257,3,1" "257,3,2147483647")
    string(REPLACE "," ";" dims "${config}")
    list(GET dims 0 n)
    list(GET dims 1 rounds)
    list(GET dims 2 divisor)
    foreach(signed 0 1)
      add_test(NAME division_${n}_${rounds}_${divisor}_${signed} COMMAND integer_division --n ${n} --rounds ${rounds} --divisor ${divisor} --signed ${signed} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(division_${n}_${rounds}_${divisor}_${signed} PROPERTIES LABELS "gpu;integer_division")
    endforeach()
  endforeach()
  foreach(rounds 1 3 4 7 8 9 31 128)
    foreach(chains 1 4)
      add_test(NAME unroll_33_${rounds}_${chains} COMMAND loop_unroll --n 33 --rounds ${rounds} --chains ${chains} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(unroll_33_${rounds}_${chains} PROPERTIES LABELS "gpu;loop_unroll")
    endforeach()
  endforeach()
  foreach(n 1 4099)
    add_test(NAME unroll_${n}_17 COMMAND loop_unroll --n ${n} --rounds 17 --chains 4 --warmup 1 --repeats 2 --iterations 3)
    set_tests_properties(unroll_${n}_17 PROPERTIES LABELS "gpu;loop_unroll")
  endforeach()
  foreach(n 1 31 32 33 63 64 65 4099)
    foreach(rounds 1 3 128)
      if(rounds EQUAL 1)
        set(body short)
      else()
        set(body loop)
      endif()
      add_test(NAME branch_${n}_${rounds} COMMAND branch_flow --n ${n} --rounds ${rounds} --body ${body} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(branch_${n}_${rounds} PROPERTIES LABELS "gpu;branch_flow")
    endforeach()
  endforeach()
endif()
