foreach(experiment reduction register_pressure)
  add_executable(${experiment} ${experiment}.cu)
  target_include_directories(${experiment} PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/generated")
  target_compile_options(${experiment} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${BPG_NVCC_OPTIONS}>)
  set_target_properties(${experiment} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
  if(BUILD_TESTING)
    add_test(NAME ${experiment}_help COMMAND ${experiment} --help)
    set_tests_properties(${experiment}_help PROPERTIES LABELS host)
  endif()
endforeach()
if(BUILD_TESTING)
  foreach(config "1,32,1" "31,32,1" "32,32,1" "33,32,1" "257,128,3" "4099,256,4" "65537,1024,1" "1048583,256,8")
    string(REPLACE "," ";" dims "${config}")
    list(GET dims 0 n)
    list(GET dims 1 block)
    list(GET dims 2 items)
    foreach(pattern mixed ones cancellation)
      add_test(NAME reduction_${n}_${block}_${items}_${pattern} COMMAND reduction --n ${n} --block ${block} --items ${items} --pattern ${pattern} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(reduction_${n}_${block}_${items}_${pattern} PROPERTIES LABELS "gpu;reduction")
    endforeach()
  endforeach()
  foreach(config "1,32" "31,32" "33,32" "257,128" "4099,256" "65536,512")
    string(REPLACE "," ";" dims "${config}")
    list(GET dims 0 n)
    list(GET dims 1 block)
    foreach(rounds 1 3 32)
      add_test(NAME register_${n}_${block}_${rounds} COMMAND register_pressure --n ${n} --block ${block} --rounds ${rounds} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(register_${n}_${block}_${rounds} PROPERTIES LABELS "gpu;register_pressure")
    endforeach()
  endforeach()
endif()
