foreach(experiment host_transfer stream_pipeline concurrent_kernels)
  add_executable(${experiment} ${experiment}.cu)
  target_include_directories(${experiment} PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/generated")
  target_compile_options(${experiment} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${BPG_NVCC_OPTIONS}>)
  set_target_properties(${experiment} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
  if(BUILD_TESTING)
    add_test(NAME ${experiment}_help COMMAND ${experiment} --help)
    set_tests_properties(${experiment}_help PROPERTIES LABELS host)
  endif()
endforeach()
if(BUILD_TESTING)
  foreach(n 1 31 33 4099 1048576)
    foreach(direction h2d d2h)
      add_test(NAME transfer_${n}_${direction} COMMAND host_transfer --n ${n} --direction ${direction} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(transfer_${n}_${direction} PROPERTIES LABELS "gpu;host_transfer")
    endforeach()
  endforeach()
  foreach(config "1,1,1,1" "33,128,3,2" "65,31,3,2" "4099,257,2,3" "4099,256,3,1" "4099,256,1,3" "1048593,65536,3,3")
    string(REPLACE "," ";" dims "${config}")
    list(GET dims 0 n)
    list(GET dims 1 chunk)
    list(GET dims 2 streams)
    list(GET dims 3 buffers)
    foreach(order interleaved phased)
      add_test(NAME pipeline_${n}_${chunk}_${streams}_${buffers}_${order} COMMAND stream_pipeline --n ${n} --chunk ${chunk} --streams ${streams} --buffers ${buffers} --order ${order} --rounds 3 --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(pipeline_${n}_${chunk}_${streams}_${buffers}_${order} PROPERTIES LABELS "gpu;stream_pipeline")
    endforeach()
  endforeach()
  foreach(config "1,1" "33,2" "65,3" "4099,4" "65537,8")
    string(REPLACE "," ";" dims "${config}")
    list(GET dims 0 n)
    list(GET dims 1 jobs)
    foreach(rounds 1 17)
      add_test(NAME concurrent_${n}_${jobs}_${rounds} COMMAND concurrent_kernels --n ${n} --jobs ${jobs} --rounds ${rounds} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(concurrent_${n}_${jobs}_${rounds} PROPERTIES LABELS "gpu;concurrent_kernels")
    endforeach()
  endforeach()
endif()
