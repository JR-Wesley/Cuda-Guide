foreach(experiment transpose cache_working_set local_array)
  add_executable(${experiment} ${experiment}.cu)
  target_include_directories(${experiment} PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/generated")
  target_compile_options(${experiment} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${BPG_NVCC_OPTIONS}>)
  set_target_properties(${experiment} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
  if(BUILD_TESTING)
    add_test(NAME ${experiment}_help COMMAND ${experiment} --help)
    set_tests_properties(${experiment}_help PROPERTIES LABELS host)
  endif()
endforeach()
if(BUILD_TESTING)
  foreach(shape "1,1" "1,33" "33,1" "31,33" "32,32" "33,31" "37,65" "1024,1024" "1023,1031")
    string(REPLACE "," ";" dims "${shape}")
    list(GET dims 0 rows)
    list(GET dims 1 cols)
    add_test(NAME transpose_${rows}x${cols} COMMAND transpose --rows ${rows} --cols ${cols} --warmup 1 --repeats 2 --iterations 3)
    set_tests_properties(transpose_${rows}x${cols} PROPERTIES LABELS "gpu;transpose")
  endforeach()
  foreach(config "1,32,32" "31,64,3" "33,32,1" "33,128,4" "4099,8192,3" "65536,1048576,32")
    string(REPLACE "," ";" dims "${config}")
    list(GET dims 0 n)
    list(GET dims 1 w)
    list(GET dims 2 steps)
    add_test(NAME cache_${n}_${w}_${steps} COMMAND cache_working_set --n ${n} --working-set ${w} --steps ${steps} --warmup 1 --repeats 2 --iterations 3)
    set_tests_properties(cache_${n}_${w}_${steps} PROPERTIES LABELS "gpu;cache_working_set")
  endforeach()
  foreach(n 1 31 33 257 4099)
    foreach(rounds 1 3 32)
      add_test(NAME local_${n}_${rounds} COMMAND local_array --n ${n} --rounds ${rounds} --warmup 1 --repeats 2 --iterations 3)
      set_tests_properties(local_${n}_${rounds} PROPERTIES LABELS "gpu;local_array")
    endforeach()
  endforeach()
endif()
